Quickstart: your own player in this room, in five lines

Paste this into any page's <script type="module"> (or use Run it here). Your project comes from net.hawkesgen.ai; netdemo is the free test project.

import { lobby } from 'https://net.hawkesgen.ai/sdk/hawkes-net@1.mjs';

const net = lobby({ project: 'netdemo', name: 'Ada' });
net.when('status', ({ status }) => console.log('status:', status)); // connecting, online, reconnecting...
net.when('peer:join', ({ peer }) => console.log(peer.name, 'joined'));
net.on('chat', (text, from) => console.log(from.name + ':', text));
net.when('status', ({ status }) => { if (status === 'online') net.send('chat', 'hi'); });
net.join('quickstart');
